1//! The Graphics Api is a trait that has all the functions that are used to work with Oden's OpenGL context.
2
3#![allow(missing_docs)]
4
5use crate::GlLoader;
6
7#[cfg_attr(feature = "mock", mockall::automock)]
8pub trait InitAndGraphicsApi {
9    fn gl_loader(&self) -> GlLoader;
10}
11
12#[cfg_attr(feature = "mock", mockall::automock)]
13pub trait GraphicsApi {
14    fn camera_gl_tex_id(&self, entity: &str, stream: i32) -> Option<u32>;
15}
16
17#[macro_export]
18#[doc(hidden)]
19macro_rules! impl_init_and_graphics_api {
20    ($interface:ident) => {
21        #[inherent::inherent]
22        impl $crate::graphics_api::InitAndGraphicsApi for $interface<'_> {
23            /// Returns the OpenGL function loader that Oden uses.
24            pub fn gl_loader(&self) -> $crate::GlLoader {
25                unsafe {
26                    if let Some(get_gl_loader) = (*self.inner).getGlLoader {
27                        get_gl_loader()
28                    } else {
29                        panic!("This version of Oden is too old to have the gl_loader function");
30                    }
31                }
32            }
33        }
34    };
35}
36
37#[macro_export]
38#[doc(hidden)]
39macro_rules! impl_graphics_api {
40    ($interface:ident) => {
41        #[inherent::inherent]
42        impl $crate::graphics_api::GraphicsApi for $interface<'_> {
43            /// Returns the OpenGL texture ID for video `stream` of `entity`.
44            ///
45            /// The OpenGL texture handle is only guaranteed to be valid this frame.
46            /// It is strongly encouraged to request a new handle every frame.
47            pub fn camera_gl_tex_id(&self, entity: &str, stream: i32) -> Option<u32> {
48                unsafe {
49                    if let Some(get_camera_gl_tex_id) = (*self.inner).getCameraGlTexId {
50                        let entity = std::ffi::CString::new(entity.trim_end_matches('\0')).unwrap();
51                        let mut tex_id_out = 0;
52                        if get_camera_gl_tex_id(
53                            entity.as_ptr(),
54                            stream,
55                            &mut tex_id_out as *mut u32,
56                        ) {
57                            Some(tex_id_out)
58                        } else {
59                            None
60                        }
61                    } else {
62                        panic!(
63                            "This version of Oden is too old to have the camera_gl_tex_id function"
64                        );
65                    }
66                }
67            }
68        }
69    };
70}
