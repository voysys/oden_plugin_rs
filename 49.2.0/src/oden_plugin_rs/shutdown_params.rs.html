6/// # ShutdownParams
7/// The [`OdenPlugin::shutdown`](crate::OdenPlugin::shutdown) call comes when the plugin is closing down.
8/// Data allocated during OdenPlugin::shutdown must be freed here.
9///
10/// ShutdownParams is a struct that exposes the Oden Plugin API functions associated with
11/// the [`OdenPlugin::shutdown`](crate::OdenPlugin::shutdown) function.
12/// The api is passed as a parameter to `shutdown(self, api: &ShutdownParams)` and the api can be accessed as shown in the example below:
13///
14/// # Examples
15/// ```ignore
16/// fn shutdown(self, api: &ShutdownParams) {
17///     api.im_draw_add_text("Hello World!",
18///         Vec3::xyz(-1.0, 0.0, -2.0),
19///         Vec3::xyz(0.0, 0.3, 0.0),
20///         Vec3::xyz(1.0, 0.0, 0.0),
21///         Vec4::rgba(0.8, 0.8, 0.8, 1.0),
22///         0,
23///     );
24/// }
25/// ```
26pub struct ShutdownParams<'a> {
27    #[doc(hidden)]
28    pub inner: *mut plugin_h::OdenPluginEntityShutdownParams,
29    #[doc(hidden)]
30    pub phantom: PhantomData<&'a i32>,
31}
32
