6#[derive(Copy, Clone)]
7enum PropertyType {
8    F32,
9    I32,
10    Bool,
11    String,
12}
13
14impl PropertyType {
15    fn new(ty: &Type) -> Self {
16        match ty {
17            Type::Path(path) => {
18                if let Some(ident) = path.path.get_ident() {
19                    match ident.to_string().as_str() {
20                        "f32" => PropertyType::F32,
21                        "i32" => PropertyType::I32,
22                        "bool" => PropertyType::Bool,
23                        "String" => PropertyType::String,
24                        _ => panic!("Unsupported property type"),
25                    }
26                } else {
27                    panic!("Unsupported property type");
28                }
29            }
30            _ => {
31                panic!("Unsupported property type");
32            }
33        }
34    }
35}

37fn get_default_value(attribute: &Attribute) -> TokenStream {
38    if let Some(TokenTree::Group(group)) = attribute.tokens.clone().into_iter().next() {
39        if group.delimiter() != Delimiter::Parenthesis {
40            return quote! { ::std::default::Default::default() };
41        }
42
43        let tokens = group.stream().into_iter().collect::<Vec<_>>();
44
45        if tokens.len() != 3 {
46            return quote! { ::std::default::Default::default() };
47        }
48
49        if let (TokenTree::Ident(ident), TokenTree::Punct(punct), TokenTree::Literal(lit)) =
50            (&tokens[0], &tokens[1], &tokens[2])
51        {
52            if ident == "default" && punct.as_char() == '=' {
53                return TokenTree::from(lit.clone()).into();
54            } else {
55                panic!("Only (default = ..) supported")
56            }
57        }
58
59        // This case is for bools with default value true/false
60        if let (TokenTree::Ident(ident), TokenTree::Punct(punct), TokenTree::Ident(default)) =
61            (&tokens[0], &tokens[1], &tokens[2])
62        {
63            if ident == "default" && punct.as_char() == '=' {
64                return TokenTree::from(default.clone()).into();
65            } else {
66                panic!("Only (default = ..) supported")
67            }
68        }
69    }
70
71    quote! { ::std::default::Default::default() }
72}

74struct Property<'a> {
75    ident: &'a Ident,
76    property_type: PropertyType,
77    default_value: TokenStream,
78}

80#[proc_macro_derive(Properties, attributes(property))]
81pub fn derive_properties(input: proc_macro::TokenStream) -> proc_macro::TokenStream {
82    let input = parse_macro_input!(input as DeriveInput);
83
84    let name = input.ident.to_string();
85    let snake_case_name = name.to_snake_case();
86
87    let mut properties = Vec::new();
88
89    let mut all_fields_are_properties = true;
90
91    if let Data::Struct(data) = &input.data {
92        if let Fields::Named(fields) = &data.fields {
93            for field in &fields.named {
94                if let Some(attribute) = field.attrs.iter().find(|a| {
95                    if let Some(ident) = a.path.get_ident() {
96                        if ident == "property" {
97                            return true;
98                        }
99                    }
100                    false
101                }) {
102                    if let Some(ident) = &field.ident {
103                        properties.push(Property {
104                            ident,
105                            property_type: PropertyType::new(&field.ty),
106                            default_value: get_default_value(attribute),
107                        });
108                    } else {
109                        all_fields_are_properties = false;
110                    }
111                } else {
112                    all_fields_are_properties = false;
113                }
114            }
115        } else {
116            panic!("Only supports Structs with named fields");
117        }
118    } else {
119        panic!("Only supports Structs");
120    }
121
122    let struct_ident = Ident::new(&name, input.ident.span());
123    let internal_mod_ident =
124        Ident::new(&format!("__{snake_case_name}_internal"), input.ident.span());
125
126    let mut load_properties = quote! {};
127    let mut save_properties = quote! {};
128    let mut gui_properties = quote! {};
129    let mut defaulted_fields = quote! {};
130
131    for property in properties.iter() {
132        let property_ident = property.ident;
133        let property_name = property.ident.to_string();
134        let property_ui_name = property_name.to_title_case();
135        let default_value = &property.default_value;
136
137        match property.property_type {
138            PropertyType::F32 => {
139                load_properties.extend(quote! {
140                    if let Some(value) = api.read_float(#property_name) {
141                        self.#property_ident = value;
142                    } else {
143                        api.write_float(#property_name, #default_value);
144                        self.#property_ident = #default_value;
145                    }
146                });
147                save_properties.extend(quote! {
148                    api.write_float(#property_name, self.#property_ident);
149                });
150                gui_properties.extend(quote! {
151                    if api.drag_float(
152                        #property_ui_name,
153                        0.01,
154                        0.0,
155                        0.0,
156                        "%.02f",
157                        1.0,
158                        &mut self.#property_ident,
159                    ) {
160                        api.write_float(#property_name, self.#property_ident);
161                    }
162                });
163                defaulted_fields.extend(quote! {
164                    #property_ident: #default_value,
165                });
166            }
167            PropertyType::I32 => {
168                load_properties.extend(quote! {
169                    if let Some(value) = api.read_int(#property_name) {
170                        self.#property_ident = value;
171                    } else {
172                        api.write_int(#property_name, #default_value);
173                        self.#property_ident = #default_value;
174                    }
175                });
176                save_properties.extend(quote! {
177                    api.write_int(#property_name, self.#property_ident);
178                });
179                gui_properties.extend(quote! {
180                    if api.drag_int(
181                        #property_ui_name,
182                        0.01,
183                        0,
184                        0,
185                        "%.0f",
186                        &mut self.#property_ident,
187                    ) {
188                        api.write_int(#property_name, self.#property_ident);
189                    }
190                });
191
192                defaulted_fields.extend(quote! {
193                    #property_ident: #default_value,
194                });
195            }
196            PropertyType::Bool => {
197                load_properties.extend(quote! {
198                    if let Some(value) = api.read_int(#property_name) {
199                        self.#property_ident = if value > 0 { true } else { false };
200                    } else {
201                        let i = if #default_value { 1 } else { 0 };
202                        api.write_int(#property_name, i);
203                        self.#property_ident = #default_value;
204                    }
205                });
206                save_properties.extend(quote! {
207                    {
208                        let i = if self.#property_ident { 1 } else { 0 };
209                        api.write_int(#property_name, i);
210                    }
211                });
212                gui_properties.extend(quote! {
213                    if api.checkbox(
214                        #property_ui_name,
215                        &mut self.#property_ident,
216                    ) {
217                        let i = if self.#property_ident { 1 } else { 0 };
218                        api.write_int(#property_name, i);
219                    }
220                });
221                defaulted_fields.extend(quote! {
222                    #property_ident: #default_value,
223                });
224            }
225            PropertyType::String => {
226                load_properties.extend(quote! {
227                    if let Some(value) = api.read_string(#property_name) {
228                        self.#property_ident = value;
229                    } else {
230                        api.write_string(#property_name, #default_value);
231                        self.#property_ident = #default_value.to_string();
232                    }
233                });
234                save_properties.extend(quote! {
235                    api.write_string(#property_name, &self.#property_ident);
236                });
237                gui_properties.extend(quote! {
238                    if api.input_text(#property_ui_name, &mut self.#property_ident) {
239                        api.write_string(#property_name, &self.#property_ident);
240                    }
241                });
242                defaulted_fields.extend(quote! {
243                    #property_ident: #default_value.to_string(),
244                });
245            }
246        }
247    }
248
249    let default_impl = if all_fields_are_properties {
250        quote! {
251            impl Default for crate::#struct_ident {
252                fn default() -> Self {
253                    Self {
254                        #defaulted_fields
255                    }
256                }
257            }
258        }
259    } else {
260        quote! {}
261    };
262
263    let expanded = quote! {
264
265        mod #internal_mod_ident {
266
267            #default_impl
268
269            impl ::oden_plugin_rs::OdenPluginWithProperties for crate::#struct_ident {
270                fn load_properties(&mut self, api: &::oden_plugin_rs::InitParams) {
271                    #load_properties
272                }
273
274                fn save_properties(&mut self, api: &::oden_plugin_rs::UpdateParams) {
275                    #save_properties
276                }
277
278                fn gui_properties(&mut self, api: &::oden_plugin_rs::GuiParams) {
279                    #gui_properties
280                }
281            }
282
283            impl crate::#struct_ident {
284                pub fn as_oden_plugin_with_properties(
285                    &mut self,
286                ) -> Option<&mut dyn ::oden_plugin_rs::OdenPluginWithProperties> {
287                    Some(self as _)
288                }
289            }
290        }
291    };
292
293    proc_macro::TokenStream::from(expanded)
294}
