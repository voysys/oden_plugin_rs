34/// An OpenGL painter using [`glow`].
35///
36/// This is responsible for painting egui and managing egui textures.
37/// You can access the underlying [`glow::Context`] with [`Self::gl`].
38///
39/// This struct must be destroyed with [`Painter::destroy`] before dropping, to ensure OpenGL
40/// objects have been properly deleted and are not leaked.
41pub struct Painter {
42    gl: Rc<glow::Context>,
43    max_texture_side: usize,
44    program: glow::Program,
45    u_screen_size: glow::UniformLocation,
46    u_sampler: glow::UniformLocation,
47    is_webgl_1: bool,
48    vao: vao::VertexArrayObject,
49    srgb_textures: bool,
50    vbo: glow::Buffer,
51    element_array_buffer: glow::Buffer,
52    textures: HashMap<egui::TextureId, glow::Texture>,
53    next_native_tex_id: u64,
54    textures_to_destroy: Vec<glow::Texture>,
55    destroyed: bool,
56}

58/// A callback function that can be used to compose an [`egui::PaintCallback`] for custom rendering
59/// with [`glow`].
60///
61/// The callback is passed, the [`egui::PaintCallbackInfo`] and the [`Painter`] which can be used to
62/// access the OpenGL context.
63///
64/// # Example
65///
66/// See the [`custom3d_glow`](https://github.com/emilk/egui/blob/master/crates/egui_demo_app/src/apps/custom3d_wgpu.rs) demo source for a detailed usage example.
67#[allow(clippy::type_complexity)]
68pub struct CallbackFn {
69    f: Box<dyn Fn(PaintCallbackInfo, &Painter) + Sync + Send>,
70}

72impl CallbackFn {
73    pub fn new<F: Fn(PaintCallbackInfo, &Painter) + Sync + Send + 'static>(callback: F) -> Self {
74        let f = Box::new(callback);
75        CallbackFn { f }
76    }

80    /// Create painter.
81    ///
82    /// Set `pp_fb_extent` to the framebuffer size to enable `sRGB` support on OpenGL ES and WebGL.
83    ///
84    /// Set `shader_prefix` if you want to turn on shader workaround e.g. `"#define APPLY_BRIGHTENING_GAMMA\n"`
85    /// (see <https://github.com/emilk/egui/issues/794>).
86    ///
87    /// # Errors
88    /// will return `Err` below cases
89    /// * failed to compile shader
90    /// * failed to create postprocess on webgl with `sRGB` support
91    /// * failed to create buffer
92    pub fn new(
93        gl: Rc<glow::Context>,
94        shader_prefix: &str,
95        shader_version: Option<ShaderVersion>,
96    ) -> Result<Painter, String> {
97        crate::check_for_gl_error_even_in_release!(&gl, "before Painter::new");
98
99        unsafe {
100            let version = gl.get_parameter_string(glow::VERSION);
101            let renderer = gl.get_parameter_string(glow::RENDERER);
102            let vendor = gl.get_parameter_string(glow::VENDOR);
103            log::debug!(
104                "\nopengl version: {version}\nopengl renderer: {renderer}\nopengl vendor: {vendor}"
105            );
106        }
107
108        #[cfg(not(target_arch = "wasm32"))]
109        if gl.version().major < 2 {
110            return Err("egui_glow requires opengl 2.0+. ".to_owned());
111        }
112
113        let max_texture_side = unsafe { gl.get_parameter_i32(glow::MAX_TEXTURE_SIZE) } as usize;
114        let shader_version = shader_version.unwrap_or_else(|| ShaderVersion::get(&gl));
115        let is_webgl_1 = shader_version == ShaderVersion::Es100;
116        let shader_version_declaration = shader_version.version_declaration();
117        log::debug!("Shader header: {:?}.", shader_version_declaration);
118
119        let supported_extensions = gl.supported_extensions();
120        log::trace!("OpenGL extensions: {supported_extensions:?}");
121        let srgb_textures = shader_version == ShaderVersion::Es300
122            || supported_extensions
123                .iter()
124                .any(|extension| extension.contains("sRGB"));
125        log::debug!("SRGB texture Support: {:?}", srgb_textures);
126
127        unsafe {
128            let vert = compile_shader(
129                &gl,
130                glow::VERTEX_SHADER,
131                &format!(
132                    "{}\n#define NEW_SHADER_INTERFACE {}\n{}\n{}",
133                    shader_version_declaration,
134                    shader_version.is_new_shader_interface() as i32,
135                    shader_prefix,
136                    VERT_SRC
137                ),
138            )?;
139            let frag = compile_shader(
140                &gl,
141                glow::FRAGMENT_SHADER,
142                &format!(
143                    "{}\n#define NEW_SHADER_INTERFACE {}\n#define SRGB_TEXTURES {}\n{}\n{}",
144                    shader_version_declaration,
145                    shader_version.is_new_shader_interface() as i32,
146                    srgb_textures as i32,
147                    shader_prefix,
148                    FRAG_SRC
149                ),
150            )?;
151            let program = link_program(&gl, [vert, frag].iter())?;
152            gl.detach_shader(program, vert);
153            gl.detach_shader(program, frag);
154            gl.delete_shader(vert);
155            gl.delete_shader(frag);
156            let u_screen_size = gl.get_uniform_location(program, "u_screen_size").unwrap();
157            let u_sampler = gl.get_uniform_location(program, "u_sampler").unwrap();
158
159            let vbo = gl.create_buffer()?;
160
161            let a_pos_loc = gl.get_attrib_location(program, "a_pos").unwrap();
162            let a_tc_loc = gl.get_attrib_location(program, "a_tc").unwrap();
163            let a_srgba_loc = gl.get_attrib_location(program, "a_srgba").unwrap();
164
165            let stride = std::mem::size_of::<Vertex>() as i32;
166            let buffer_infos = vec![
167                vao::BufferInfo {
168                    location: a_pos_loc,
169                    vector_size: 2,
170                    data_type: glow::FLOAT,
171                    normalized: false,
172                    stride,
173                    offset: offset_of!(Vertex, pos) as i32,
174                },
175                vao::BufferInfo {
176                    location: a_tc_loc,
177                    vector_size: 2,
178                    data_type: glow::FLOAT,
179                    normalized: false,
180                    stride,
181                    offset: offset_of!(Vertex, uv) as i32,
182                },
183                vao::BufferInfo {
184                    location: a_srgba_loc,
185                    vector_size: 4,
186                    data_type: glow::UNSIGNED_BYTE,
187                    normalized: false,
188                    stride,
189                    offset: offset_of!(Vertex, color) as i32,
190                },
191            ];
192            let vao = vao::VertexArrayObject::new(&gl, vbo, buffer_infos);
193
194            let element_array_buffer = gl.create_buffer()?;
195
196            crate::check_for_gl_error_even_in_release!(&gl, "after Painter::new");
197
198            Ok(Painter {
199                gl,
200                max_texture_side,
201                program,
202                u_screen_size,
203                u_sampler,
204                is_webgl_1,
205                vao,
206                srgb_textures,
207                vbo,
208                element_array_buffer,
209                textures: Default::default(),
210                next_native_tex_id: 1 << 32,
211                textures_to_destroy: Vec::new(),
212                destroyed: false,
213            })
214        }
215    }

241    unsafe fn prepare_painting(
242        &mut self,
243        [width_in_pixels, height_in_pixels]: [u32; 2],
244        pixels_per_point: f32,
245    ) {
246        self.gl.enable(glow::SCISSOR_TEST);
247        self.gl.disable(glow::CULL_FACE);
248        self.gl.disable(glow::DEPTH_TEST);
249
250        self.gl.color_mask(true, true, true, true);
251
252        self.gl.enable(glow::BLEND);
253        self.gl
254            .blend_equation_separate(glow::FUNC_ADD, glow::FUNC_ADD);
255        self.gl.blend_func_separate(
256            glow::ONE,
257            glow::ONE_MINUS_SRC_ALPHA,
258            glow::ONE_MINUS_DST_ALPHA,
259            glow::ONE,
260        );
261
262        if !cfg!(target_arch = "wasm32") {
263            self.gl.disable(glow::FRAMEBUFFER_SRGB);
264            check_for_gl_error!(&self.gl, "FRAMEBUFFER_SRGB");
265        }
266
267        let width_in_points = width_in_pixels as f32 / pixels_per_point;
268        let height_in_points = height_in_pixels as f32 / pixels_per_point;
269
270        self.gl
271            .viewport(0, 0, width_in_pixels as i32, height_in_pixels as i32);
272        self.gl.use_program(Some(self.program));
273
274        self.gl
275            .uniform_2_f32(Some(&self.u_screen_size), width_in_points, height_in_points);
276        self.gl.uniform_1_i32(Some(&self.u_sampler), 0);
277        self.gl.active_texture(glow::TEXTURE0);
278
279        self.vao.bind(&self.gl);
280        self.gl
281            .bind_buffer(glow::ELEMENT_ARRAY_BUFFER, Some(self.element_array_buffer));
282
283        check_for_gl_error!(&self.gl, "prepare_painting");
284    }

286    /// You are expected to have cleared the color buffer before calling this.
287    pub fn paint_and_update_textures(
288        &mut self,
289        screen_size_px: [u32; 2],
290        pixels_per_point: f32,
291        clipped_primitives: &[egui::ClippedPrimitive],
292        textures_delta: &egui::TexturesDelta,
293    ) {
294        for (id, image_delta) in &textures_delta.set {
295            self.set_texture(*id, image_delta);
296        }
297
298        self.paint_primitives(screen_size_px, pixels_per_point, clipped_primitives);
299
300        for &id in &textures_delta.free {
301            self.free_texture(id);
302        }
303    }

305    /// Main entry-point for painting a frame.
306    ///
307    /// You should call `target.clear_color(..)` before
308    /// and `target.finish()` after this.
309    ///
310    /// The following OpenGL features will be set:
311    /// - Scissor test will be enabled
312    /// - Cull face will be disabled
313    /// - Blend will be enabled
314    ///
315    /// The scissor area and blend parameters will be changed.
316    ///
317    /// As well as this, the following objects will be unset:
318    /// - Vertex Buffer
319    /// - Element Buffer
320    /// - Texture (and active texture will be set to 0)
321    /// - Program
322    ///
323    /// Please be mindful of these effects when integrating into your program, and also be mindful
324    /// of the effects your program might have on this code. Look at the source if in doubt.
325    pub fn paint_primitives(
326        &mut self,
327        screen_size_px: [u32; 2],
328        pixels_per_point: f32,
329        clipped_primitives: &[egui::ClippedPrimitive],
330    ) {
331        self.assert_not_destroyed();
332
333        unsafe { self.prepare_painting(screen_size_px, pixels_per_point) };
334
335        for egui::ClippedPrimitive {
336            clip_rect,
337            primitive,
338        } in clipped_primitives
339        {
340            set_clip_rect(&self.gl, screen_size_px, pixels_per_point, *clip_rect);
341
342            match primitive {
343                Primitive::Mesh(mesh) => {
344                    self.paint_mesh(mesh);
345                }
346                Primitive::Callback(callback) => {
347                    if callback.rect.is_positive() {
348                        // Transform callback rect to physical pixels:
349                        let rect_min_x = pixels_per_point * callback.rect.min.x;
350                        let rect_min_y = pixels_per_point * callback.rect.min.y;
351                        let rect_max_x = pixels_per_point * callback.rect.max.x;
352                        let rect_max_y = pixels_per_point * callback.rect.max.y;
353
354                        let rect_min_x = rect_min_x.round() as i32;
355                        let rect_min_y = rect_min_y.round() as i32;
356                        let rect_max_x = rect_max_x.round() as i32;
357                        let rect_max_y = rect_max_y.round() as i32;
358
359                        unsafe {
360                            self.gl.viewport(
361                                0,
362                                0,
363                                rect_max_x - rect_min_x,
364                                rect_max_y - rect_min_y,
365                            );
366                        }
367
368                        let info = egui::PaintCallbackInfo {
369                            viewport: callback.rect,
370                            clip_rect: *clip_rect,
371                            pixels_per_point,
372                            screen_size_px,
373                        };
374
375                        if let Some(callback) = callback.callback.downcast_ref::<CallbackFn>() {
376                            (callback.f)(info, self);
377                        } else {
378                            log::warn!("Warning: Unsupported render callback. Expected egui_glow::CallbackFn");
379                        }
380
381                        check_for_gl_error!(&self.gl, "callback");
382
383                        unsafe { self.prepare_painting(screen_size_px, pixels_per_point) };
384                    }
385                }
386            }
387        }
388
389        unsafe {
390            self.vao.unbind(&self.gl);
391            self.gl.bind_buffer(glow::ELEMENT_ARRAY_BUFFER, None);
392
393            self.gl.disable(glow::SCISSOR_TEST);
394
395            self.gl.enable(glow::DEPTH_TEST);
396            self.gl.disable(glow::CULL_FACE);
397            self.gl.disable(glow::BLEND);
398
399            check_for_gl_error!(&self.gl, "painting");
400        }
401    }

403    #[inline(never)]
404    fn paint_mesh(&mut self, mesh: &Mesh) {
405        debug_assert!(mesh.is_valid());
406        if let Some(texture) = self.texture(mesh.texture_id) {
407            unsafe {
408                self.gl.bind_buffer(glow::ARRAY_BUFFER, Some(self.vbo));
409                self.gl.buffer_data_u8_slice(
410                    glow::ARRAY_BUFFER,
411                    bytemuck::cast_slice(&mesh.vertices),
412                    glow::STREAM_DRAW,
413                );
414
415                self.gl
416                    .bind_buffer(glow::ELEMENT_ARRAY_BUFFER, Some(self.element_array_buffer));
417                self.gl.buffer_data_u8_slice(
418                    glow::ELEMENT_ARRAY_BUFFER,
419                    bytemuck::cast_slice(&mesh.indices),
420                    glow::STREAM_DRAW,
421                );
422
423                self.gl.bind_texture(glow::TEXTURE_2D, Some(texture));
424            }
425
426            unsafe {
427                self.gl.draw_elements(
428                    glow::TRIANGLES,
429                    mesh.indices.len() as i32,
430                    glow::UNSIGNED_INT,
431                    0,
432                );
433            }
434
435            check_for_gl_error!(&self.gl, "paint_mesh");
436        } else {
437            log::warn!("Failed to find texture {:?}", mesh.texture_id);
438        }
439    }

441    pub fn set_texture(&mut self, tex_id: egui::TextureId, delta: &egui::epaint::ImageDelta) {
442        self.assert_not_destroyed();
443
444        let glow_texture = *self
445            .textures
446            .entry(tex_id)
447            .or_insert_with(|| unsafe { self.gl.create_texture().unwrap() });
448        unsafe {
449            self.gl.bind_texture(glow::TEXTURE_2D, Some(glow_texture));
450        }
451
452        match &delta.image {
453            egui::ImageData::Color(image) => {
454                assert_eq!(
455                    image.width() * image.height(),
456                    image.pixels.len(),
457                    "Mismatch between texture size and texel count"
458                );
459
460                let data: &[u8] = bytemuck::cast_slice(image.pixels.as_ref());
461
462                self.upload_texture_srgb(delta.pos, image.size, delta.options, data);
463            }
464            egui::ImageData::Font(image) => {
465                assert_eq!(
466                    image.width() * image.height(),
467                    image.pixels.len(),
468                    "Mismatch between texture size and texel count"
469                );
470
471                let data: Vec<u8> = image
472                    .srgba_pixels(None)
473                    .flat_map(|a| a.to_array())
474                    .collect();
475
476                self.upload_texture_srgb(delta.pos, image.size, delta.options, &data);
477            }
478        };
479    }

481    fn upload_texture_srgb(
482        &mut self,
483        pos: Option<[usize; 2]>,
484        [w, h]: [usize; 2],
485        options: egui::TextureOptions,
486        data: &[u8],
487    ) {
488        assert_eq!(data.len(), w * h * 4);
489        assert!(
490            w <= self.max_texture_side && h <= self.max_texture_side,
491            "Got a texture image of size {}x{}, but the maximum supported texture side is only {}",
492            w,
493            h,
494            self.max_texture_side
495        );
496
497        unsafe {
498            self.gl.tex_parameter_i32(
499                glow::TEXTURE_2D,
500                glow::TEXTURE_MAG_FILTER,
501                options.magnification.glow_code() as i32,
502            );
503            self.gl.tex_parameter_i32(
504                glow::TEXTURE_2D,
505                glow::TEXTURE_MIN_FILTER,
506                options.minification.glow_code() as i32,
507            );
508
509            self.gl.tex_parameter_i32(
510                glow::TEXTURE_2D,
511                glow::TEXTURE_WRAP_S,
512                glow::CLAMP_TO_EDGE as i32,
513            );
514            self.gl.tex_parameter_i32(
515                glow::TEXTURE_2D,
516                glow::TEXTURE_WRAP_T,
517                glow::CLAMP_TO_EDGE as i32,
518            );
519            check_for_gl_error!(&self.gl, "tex_parameter");
520
521            let (internal_format, src_format) = if self.is_webgl_1 {
522                let format = if self.srgb_textures {
523                    glow::SRGB_ALPHA
524                } else {
525                    glow::RGBA
526                };
527                (format, format)
528            } else if self.srgb_textures {
529                (glow::SRGB8_ALPHA8, glow::RGBA)
530            } else {
531                (glow::RGBA8, glow::RGBA)
532            };
533
534            self.gl.pixel_store_i32(glow::UNPACK_ALIGNMENT, 1);
535
536            let level = 0;
537            if let Some([x, y]) = pos {
538                self.gl.tex_sub_image_2d(
539                    glow::TEXTURE_2D,
540                    level,
541                    x as _,
542                    y as _,
543                    w as _,
544                    h as _,
545                    src_format,
546                    glow::UNSIGNED_BYTE,
547                    glow::PixelUnpackData::Slice(data),
548                );
549                check_for_gl_error!(&self.gl, "tex_sub_image_2d");
550            } else {
551                let border = 0;
552                self.gl.tex_image_2d(
553                    glow::TEXTURE_2D,
554                    level,
555                    internal_format as _,
556                    w as _,
557                    h as _,
558                    border,
559                    src_format,
560                    glow::UNSIGNED_BYTE,
561                    Some(data),
562                );
563                check_for_gl_error!(&self.gl, "tex_image_2d");
564            }
565        }
566    }

631    /// This function must be called before [`Painter`] is dropped, as [`Painter`] has some OpenGL objects
632    /// that should be deleted.
633    unsafe fn destroy_gl(&self) {
634        self.gl.delete_program(self.program);
635        for tex in self.textures.values() {
636            self.gl.delete_texture(*tex);
637        }
638        self.gl.delete_buffer(self.vbo);
639        self.gl.delete_buffer(self.element_array_buffer);
640        for t in &self.textures_to_destroy {
641            self.gl.delete_texture(*t);
642        }
643    }
644
645    pub fn destroy(&mut self) {
646        if !self.destroyed {
647            unsafe {
648                self.destroy_gl();
649            }
650            self.destroyed = true;
651        }
652    }

659pub fn clear(gl: &glow::Context, screen_size_in_pixels: [u32; 2], clear_color: [f32; 4]) {
660    unsafe {
661        gl.disable(glow::SCISSOR_TEST);
662
663        gl.viewport(
664            0,
665            0,
666            screen_size_in_pixels[0] as i32,
667            screen_size_in_pixels[1] as i32,
668        );
669        gl.clear_color(
670            clear_color[0],
671            clear_color[1],
672            clear_color[2],
673            clear_color[3],
674        );
675        gl.clear(glow::COLOR_BUFFER_BIT);
676    }
677}

689fn set_clip_rect(
690    gl: &glow::Context,
691    [width_px, height_px]: [u32; 2],
692    pixels_per_point: f32,
693    clip_rect: Rect,
694) {
695    // Transform clip rect to physical pixels:
696    let clip_min_x = pixels_per_point * clip_rect.min.x;
697    let clip_min_y = pixels_per_point * clip_rect.min.y;
698    let clip_max_x = pixels_per_point * clip_rect.max.x;
699    let clip_max_y = pixels_per_point * clip_rect.max.y;
700
701    // Round to integer:
702    let clip_min_x = clip_min_x.round() as i32;
703    let clip_min_y = clip_min_y.round() as i32;
704    let clip_max_x = clip_max_x.round() as i32;
705    let clip_max_y = clip_max_y.round() as i32;
706
707    // Clamp:
708    let clip_min_x = clip_min_x.clamp(0, width_px as i32);
709    let clip_min_y = clip_min_y.clamp(0, height_px as i32);
710    let clip_max_x = clip_max_x.clamp(clip_min_x, width_px as i32);
711    let clip_max_y = clip_max_y.clamp(clip_min_y, height_px as i32);
712
713    unsafe {
714        gl.scissor(
715            clip_min_x,
716            height_px as i32 - clip_max_y,
717            clip_max_x - clip_min_x,
718            clip_max_y - clip_min_y,
719        );
720    }
721}
