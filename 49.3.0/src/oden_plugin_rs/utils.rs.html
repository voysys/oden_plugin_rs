7pub(crate) fn utf8_from_raw(buf: &[u8]) -> Result<String, std::string::FromUtf8Error> {
8    // Filter buf from any '\0' before creating string as it seems to cause panics
9    // at some print/log calls.
10    if let Some(p) = buf.iter().position(|&v| v == 0) {
11        String::from_utf8(buf[..p].to_vec())
12    } else {
13        Ok(String::new())
14    }
15}

17// For performance reasons we don't want to use `utf8_from_raw` for large strings
18// We use this for settings strings and scene strings as they should always be valid utf-8
19pub(crate) fn utf8_from_raw_to_first_null_terminator(
20    buf: &[u8],
21) -> Result<String, std::string::FromUtf8Error> {
22    // Find the position of the first null terminator
23    let end_pos = buf.iter().position(|&b| b == 0).unwrap_or(buf.len());
24
25    let valid_bytes = &buf[..end_pos];
26
27    String::from_utf8(valid_bytes.to_vec())
28}

37/// Converts a point in space to camera coordinate for a video in a stitched video.
38/// This is useful if the user want be able to press a pixel coordinate on screen and some
39/// operation should be done on that pixel in the camera image.
40/// Example:
41/// We have a stitched video named "My Stitched video entity" with 3 videos(Stream 1, 2 and 3).
42/// When the user presses somewhere in the stitched video we will display which pixels in the original video that the mouse was over.
43/// ```no_run
44/// # fn example(api: &oden_plugin_rs::UpdateParams) {
45///     if api.mouse_pressed_since_last_swap(){
46///         let mouse_ray = api.mouse_ray();
47///         for i in 1..4 {
48///             if let (Some(calibration), Some(video_size), Some(crop)) = (
49///                 api.camera_calibration("My Stitched Video Entity", i),
50///                 api.entity_video_size("My Stitched Video Entity", i),
51///                 api.camera_crop("My Stitched Video Entity", i),
52///             ) {
53///                 if let Some(result) = oden_plugin_rs::utils::reproject_point_to_camera_coords(
54///                     mouse_ray.direction,
55///                     calibration,
56///                     crop,
57///                     video_size,
58///                 ) {
59///                     oden_plugin_rs::log::info!("User pressed video {} camera pixel {} {}", i, result.pixel_coords.x(), result.pixel_coords.y());
60///                 }
61///             }
62///         }
63///     }
64/// # }
65/// ```
66#[cfg(feature = "glam_conversion")]
67pub fn reproject_point_to_camera_coords(
68    point_to_reproject: Vec3,
69    calibration: OdenCameraCalibration,
70    crop: OdenCameraCropData,
71    resolution: Vec2i,
72) -> Option<ReprojectionResult> {
73    let mut point: glam::Vec3 = point_to_reproject.into();
74    let width = resolution.x();
75    let height = resolution.y();
76
77    let camera_rotation = glam::Quat::from_array(calibration.rotation);
78    let global_position = glam::Vec3::from_array(calibration.globalPosition);
79    let global_rotation = glam::Quat::from_array(calibration.globalRotation);
80
81    {
82        point = point - global_rotation.mul_vec3(global_position) + global_position;
83        point = ((global_rotation * camera_rotation).inverse()).mul_vec3(point);
84
85        // point is now in camera local coords
86
87        let aspect = width as f32 / height as f32;
88
89        let mut p_norm = point.length();
90        p_norm = p_norm.max(0.00000000001);
91
92        let theta = (-point.z / p_norm).acos();
93
94        let r = calibration.k[0] * theta
95            + calibration.k[1] * theta * theta
96            + calibration.k[2] * theta.powi(3)
97            + calibration.k[3] * theta.powi(4)
98            + calibration.k[4] * theta.powi(5);
99
100        let image_unit = glam::Vec2::new(point.x, point.y).normalize();
101
102        let mut pos = r * image_unit;
103
104        let to_lower_left_crop_edge =
105            glam::Vec2::new(2.0 * aspect * crop.left - aspect, 2.0 * crop.down - 1.0);
106        let after_crop_center_to_top_right_corner = glam::Vec2::new(
107            aspect * (1.0 - crop.left - crop.right),
108            1.0 - crop.down - crop.up,
109        );
110        let full_crop_distance = after_crop_center_to_top_right_corner.length();
111        let center_of_crop = to_lower_left_crop_edge + after_crop_center_to_top_right_corner;
112        let crop_distance = (pos - center_of_crop).length();
113        let circular_crop_diff =
114            (full_crop_distance - crop_distance) / full_crop_distance - crop.circular;
115
116        pos += glam::Vec2::new(calibration.offset[0], calibration.offset[1]);
117
118        pos.x /= aspect;
119
120        let uv = pos * 0.5 + glam::Vec2::new(0.5, 0.5);
121
122        let theta_diff = calibration.maxTheta - theta;
123
124        let uv_coords = glam::Vec2::new(uv.x * aspect, 1.0 - uv.y);
125        let pixel_coords = glam::Vec2::new(uv.x * width as f32, (1.0 - uv.y) * height as f32);
126
127        if theta_diff < 0.0
128            || uv.x < 0.0 + crop.left
129            || uv.x > 1.0 - crop.right
130            || uv.y < 0.0 + crop.down
131            || uv.y > 1.0 - crop.up
132            || circular_crop_diff < 0.0
133        {
134            return None;
135        }
136
137        Some(ReprojectionResult {
138            uv_coords: uv_coords.into(),
139            pixel_coords: pixel_coords.into(),
140        })
141    }
142}
