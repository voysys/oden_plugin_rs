14pub struct ScreenSpaceGuard<'a> {
15    #[doc(hidden)]
16    pub(crate) inner: *mut crate::plugin_h::OdenPluginEntityUpdateParams,
17    #[doc(hidden)]
18    pub(crate) phantom: PhantomData<&'a i32>,
19}
20
21impl Default for ScreenSpaceGuard<'_> {
22    fn default() -> Self {
23        Self {
24            inner: std::ptr::null_mut(),
25            phantom: PhantomData,
26        }
27    }
28}
29
30impl Drop for ScreenSpaceGuard<'_> {
31    fn drop(&mut self) {
32        if !self.inner.is_null() {
33            unsafe {
34                if let Some(screen_space_pop) = (*self.inner).imDrawPopScreenSpace {
35                    screen_space_pop()
36                }
37            }
38        }
39    }
40}
