1//! Host side of the message bus.
2//!
3//! The host publishes a *factory record* — the address of a [`BusHostVTable`]
4//! living in the host executable — under [`HOST_FACTORY_KEY`] in the plugin
5//! data-sharing store, before any plugin runs.
6//! [`BusOwner::open`](super::BusOwner::open) allocates every bus through
7//! that factory, so both the bus vtable code and the `BusState` reside in the
8//! host binary and stay valid across any plugin DLL/SO unload.
9//!
10//! Oden does this by linking this crate into the host (via `oden_rs`) and calling
11//! [`host_factory_record`] there; the record's function pointers therefore point
12//! into the host's own copy of `imp`, not into any plugin.
13
14use std::ffi::c_void;
15
16use super::{
17    abi::{BusHandle, BusHostVTable},
18    imp,
19};
20use crate::SettingsApi;
21
22/// Data-sharing key under which the host publishes its bus factory record.
23pub const HOST_FACTORY_KEY: &str = "oden_message_bus.host";
24
25/// Magic prefix on the factory record. Bump if the layout of [`BusHostVTable`],
26/// the record, or the [`BusVTable`](super::abi::BusVTable) behind the handles it
27/// creates ever needs an incompatible change post-merge.
28const HOST_FACTORY_MAGIC: [u8; 8] = *b"ODENBUSF";
29
30/// Size in bytes of the on-wire factory record. 8 (magic) + 1 (pointer width) +
31/// 7 (reserved) + 8 (vtable) + 8 (reserved) = 32, mirroring the bus discovery record.
32pub const HOST_FACTORY_RECORD_SIZE: usize = 32;
33
34unsafe extern "C" fn create_bus(out_handle: *mut BusHandle) -> bool {
35    if out_handle.is_null() {
36        return false;
37    }
38    unsafe { *out_handle = imp::create_bus_handle() };
39    true
40}
41
42static HOST_VTABLE: BusHostVTable = BusHostVTable { create_bus };
43
44/// Serialize the factory record for this build's [`BusHostVTable`]. The record
45/// embeds pointers into whatever binary this crate is linked into, so it must
46/// only be published by a module that is never unloaded — in practice the host
47/// executable.
48pub fn host_factory_record() -> [u8; HOST_FACTORY_RECORD_SIZE] {
49    let mut out = [0u8; HOST_FACTORY_RECORD_SIZE];
50    out[0..8].copy_from_slice(&HOST_FACTORY_MAGIC);
51    out[8] = std::mem::size_of::<usize>() as u8;
52    let vtable_addr = &HOST_VTABLE as *const BusHostVTable as usize as u64;
53    out[16..24].copy_from_slice(&vtable_addr.to_le_bytes());
54    out
55}
56
57fn read_host_factory_record(bytes: &[u8]) -> Option<*const BusHostVTable> {
58    if bytes.len() != HOST_FACTORY_RECORD_SIZE {
59        return None;
60    }
61    if bytes[0..8] != HOST_FACTORY_MAGIC {
62        return None;
63    }
64    if bytes[8] as usize != std::mem::size_of::<usize>() {
65        return None;
66    }
67    let vtable_addr = u64::from_le_bytes(bytes[16..24].try_into().ok()?);
68    if vtable_addr == 0 {
69        return None;
70    }
71    Some(vtable_addr as usize as *const BusHostVTable)
72}
73
74/// Allocate a fresh bus through the host's factory. Returns `None` if this host
75/// publishes no factory record (or an invalid one).
76pub(crate) fn create_bus_via_host(api: &dyn SettingsApi) -> Option<BusHandle> {
77    let (data, _) = api
78        .query_data(HOST_FACTORY_KEY, HOST_FACTORY_RECORD_SIZE as i32)
79        .ok()?;
80    let vtable = read_host_factory_record(&data)?;
81    let mut handle = BusHandle {
82        vtable: std::ptr::null(),
83        this: std::ptr::null_mut::<c_void>(),
84    };
85    // SAFETY: same trust model as bus discovery records — this key is only ever
86    // written by the host running this module's `host_factory_record`, and the
87    // vtable it points at lives in the host executable for the process lifetime.
88    if !unsafe { ((*vtable).create_bus)(&mut handle as *mut BusHandle) } {
89        return None;
90    }
91    if handle.vtable.is_null() || handle.this.is_null() {
92        return None;
93    }
94    Some(handle)
95}
