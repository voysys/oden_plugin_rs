1//! ABI for the cross-plugin message bus.
2//!
3//! Everything in this module is `#[repr(C)]` and crosses the plugin boundary. It must
4//! stay stable so plugins built with different toolchains can interoperate. See the
5//! `message_bus` module-level documentation for the design rationale and buffer-ownership
6//! rules.
7
8#![allow(missing_docs)]
9
10use std::ffi::c_void;
11
12/// Borrowed view of a UTF-8 string. Input-only; the callee copies before returning.
13#[repr(C)]
14#[derive(Copy, Clone)]
15pub struct BusStr {
16    pub ptr: *const u8,
17    pub len: usize,
18}
19
20/// Borrowed view of opaque bytes. Input-only; the callee copies before returning.
21#[repr(C)]
22#[derive(Copy, Clone)]
23pub struct BusBytes {
24    pub ptr: *const u8,
25    pub len: usize,
26}
27
28/// Opaque subscription handle returned by `subscribe` and consumed by `unsubscribe`/recvs.
29pub type BusSubHandle = u64;
30
31/// Status code returned by every vtable function. A `#[repr(transparent)]`
32/// wrapper over `u32` rather than a Rust enum so that a code minted by a build
33/// with newer status values is an ordinary value to receive — not an invalid
34/// enum discriminant, which would be UB at the call site. Callers must treat
35/// unknown codes as errors.
36#[repr(transparent)]
37#[derive(Copy, Clone, PartialEq, Eq)]
38pub struct BusStatus(pub u32);
39
40impl BusStatus {
41    pub const OK: Self = Self(0);
42    pub const EMPTY: Self = Self(1);
43    pub const INVALID_HANDLE: Self = Self(2);
44    pub const BUFFER_TOO_SMALL: Self = Self(3);
45    pub const QUEUE_FULL: Self = Self(4);
46}
47
48const _: () = assert!(std::mem::size_of::<BusStatus>() == 4);
49
50impl std::fmt::Debug for BusStatus {
51    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
52        match *self {
53            Self::OK => f.write_str("BusStatus::OK"),
54            Self::EMPTY => f.write_str("BusStatus::EMPTY"),
55            Self::INVALID_HANDLE => f.write_str("BusStatus::INVALID_HANDLE"),
56            Self::BUFFER_TOO_SMALL => f.write_str("BusStatus::BUFFER_TOO_SMALL"),
57            Self::QUEUE_FULL => f.write_str("BusStatus::QUEUE_FULL"),
58            Self(other) => write!(f, "BusStatus({other})"),
59        }
60    }
61}

63/// The thin client-side handle. Pairs a `vtable` pointer with the owner-allocated `this`.
64#[repr(C)]
65#[derive(Copy, Clone)]
66pub struct BusHandle {
67    pub vtable: *const BusVTable,
68    pub this: *mut c_void,
69}
70
71unsafe impl Send for BusHandle {}
72unsafe impl Sync for BusHandle {}
73
74/// vtable behind the bus factory record published by the host (see the `host` module).
75/// Its function pointers live in the host executable, which is never unloaded, so a
76/// bus allocated through it survives any plugin DLL unload.
77#[repr(C)]
78pub struct BusHostVTable {
79    /// Allocate a fresh bus in the publisher's memory. The returned handle has
80    /// refcount 1.
81    /// Returns `false` (leaving `*out_handle` untouched) on failure.
82    pub create_bus: unsafe extern "C" fn(out_handle: *mut BusHandle) -> bool,
83}
84
85/// vtable for a live bus. Buses are only allocated through [`BusHostVTable`], so
86/// these function pointers run in the host executable and its allocator owns
87/// every internal allocation.
88///
89/// All plugins in a build share the same source for this struct, so the layout is
90/// identical across them. There is no version field — any incompatible change
91/// requires bumping the magic constant in the discovery record and rebuilding
92/// every plugin. Adding a `BusStatus` code is not such a change: status is a
93/// plain `u32` on the wire and callers treat unknown codes as errors.
94#[repr(C)]
95pub struct BusVTable {
96    /// Increment the bus refcount. Called when a handle (`BusOwner`, `BusClient`)
97    /// adopts a new reference to the bus.
98    pub retain: unsafe extern "C" fn(this: *mut c_void),
99
100    /// Decrement the bus refcount. The bus is never freed (the `BusState` is leaked,
101    /// see the impl), so this only keeps the advisory count balanced.
102    pub release: unsafe extern "C" fn(this: *mut c_void),
103
104    /// Client → owner. Caller's `name`/`payload` are borrowed; callee copies before
105    /// returning. The upstream queue is bounded; messages sent while no owner is
106    /// draining wait in the queue for the next owner.
107    pub send_upstream:
108        unsafe extern "C" fn(this: *mut c_void, name: BusStr, payload: BusBytes) -> BusStatus,
109
110    /// Owner → matching subscribers. Same borrow rules as `send_upstream`. A
111    /// broadcast that matches no live subscription is parked (bounded) and
112    /// delivered to the first later subscription whose name filter matches.
113    pub try_broadcast:
114        unsafe extern "C" fn(this: *mut c_void, name: BusStr, payload: BusBytes) -> BusStatus,
115
116    /// Register a subscription. `names` is a borrowed array of `BusStr`; callee copies.
117    /// `queue_cap` of 0 means "use the default".
118    pub subscribe: unsafe extern "C" fn(
119        this: *mut c_void,
120        names: *const BusStr,
121        count: usize,
122        queue_cap: usize,
123        out_handle: *mut BusSubHandle,
124    ) -> BusStatus,
125
126    /// Drop a subscription registered by `subscribe`.
127    pub unsubscribe: unsafe extern "C" fn(this: *mut c_void, handle: BusSubHandle) -> BusStatus,
128
129    /// Receive on a client subscription without waiting for a message. Not
130    /// wait-free: blocks on a per-subscription lock held for the full wait of
131    /// any concurrent `recv_downstream_timeout` on the same subscription.
132    pub try_recv_downstream: unsafe extern "C" fn(
133        this: *mut c_void,
134        sub: BusSubHandle,
135        name_buf: *mut u8,
136        name_cap: usize,
137        name_len_out: *mut usize,
138        payload_buf: *mut u8,
139        payload_cap: usize,
140        payload_len_out: *mut usize,
141    ) -> BusStatus,
142
143    /// Blocking receive on a client subscription.
144    /// `timeout_ns == 0` is non-blocking, `u64::MAX` blocks forever.
145    pub recv_downstream_timeout: unsafe extern "C" fn(
146        this: *mut c_void,
147        sub: BusSubHandle,
148        timeout_ns: u64,
149        name_buf: *mut u8,
150        name_cap: usize,
151        name_len_out: *mut usize,
152        payload_buf: *mut u8,
153        payload_cap: usize,
154        payload_len_out: *mut usize,
155    ) -> BusStatus,
156
157    /// Receive on the upstream lane (owner draining client sends) without
158    /// waiting for a message. Not wait-free: blocks on the upstream lock held
159    /// for the full wait of any concurrent `recv_upstream_timeout`.
160    pub try_recv_upstream: unsafe extern "C" fn(
161        this: *mut c_void,
162        name_buf: *mut u8,
163        name_cap: usize,
164        name_len_out: *mut usize,
165        payload_buf: *mut u8,
166        payload_cap: usize,
167        payload_len_out: *mut usize,
168    ) -> BusStatus,
169
170    /// Blocking receive on the upstream lane.
171    pub recv_upstream_timeout: unsafe extern "C" fn(
172        this: *mut c_void,
173        timeout_ns: u64,
174        name_buf: *mut u8,
175        name_cap: usize,
176        name_len_out: *mut usize,
177        payload_buf: *mut u8,
178        payload_cap: usize,
179        payload_len_out: *mut usize,
180    ) -> BusStatus,
181
182    /// Register a one-shot wake callback for one lane. `sub == 0` targets the
183    /// upstream lane; any non-zero value targets the matching client subscription.
184    /// A lane holds any number of concurrent registrations; `*out_token` receives
185    /// a token identifying this one for `unregister_waker`.
186    ///
187    /// Ownership of `wake_ctx` transfers to the bus only when this returns `Ok`.
188    /// On any other status the bus has not touched `wake_ctx` and the caller
189    /// still owns it. After `Ok`, exactly one of the following releases the
190    /// bus's reference: a matching send or removal of the subscription fires
191    /// `wake_fn(wake_ctx)` (which takes ownership), or `unregister_waker`
192    /// removes the entry and calls `drop_ctx(wake_ctx)`.
193    /// Each registration fires at most once;
194    /// re-register after a fire to keep waiting.
195    ///
196    /// The bus releases its internal locks before invoking `wake_fn`, so
197    /// `wake_fn` may safely call back into the bus — no deadlock and no UAF on
198    /// the ctx pointer.
199    pub register_waker: unsafe extern "C" fn(
200        this: *mut c_void,
201        sub: BusSubHandle,
202        wake_fn: unsafe extern "C" fn(ctx: *mut c_void),
203        drop_ctx: unsafe extern "C" fn(ctx: *mut c_void),
204        wake_ctx: *mut c_void,
205        out_token: *mut u64,
206    ) -> BusStatus,
207
208    /// Remove a waker registration by token, releasing the stored ctx via
209    /// `drop_ctx`. Returns `InvalidHandle` if the token is unknown — it already
210    /// fired, was already unregistered, or its subscription was dropped; in every
211    /// such case the bus has already released the registration's ctx.
212    pub unregister_waker:
213        unsafe extern "C" fn(this: *mut c_void, sub: BusSubHandle, token: u64) -> BusStatus,
214}
