9/// # DrawParams
10/// The function OdenPlugin::draw allows for using raw OpenGL commands.
11/// > IMPORTANT NOTE: This is very complex and is not covered in the example plugin.
12///
13/// > NOTE: The hello_world.rs plugin does not implement the draw function by default since it is an very complex feature.
14/// > If desired the function below can be added to the `OdenPlugin` implementation for `State`.
15///
16/// DrawParams is a struct that exposes the Oden Plugin API functions associated with
17/// the OdenPlugin::draw function.
18/// The API is passed as a parameter to `draw(&mut self, api: &DrawParams)` and is accessed as shown in the example below:
19///
20/// # Examples
21/// ```ignore
22/// fn draw(&mut self, api: &DrawParams) {
23///     api.gl_loader();
24/// }
25/// ```
26pub struct DrawParams<'a> {
27    #[doc(hidden)]
28    pub inner: *mut crate::plugin_h::OdenPluginEntityDrawParams,
29    #[doc(hidden)]
30    pub recording_playback_channel: &'a Receiver<Vec<u8>>,
31}

41#[inherent::inherent]
42impl DrawApi for DrawParams<'_> {
43    /// Returns the plugin entity world transformation matrix.
44    ///
45    /// Drawing using e.g. ImDraw is made with an identity world matrix.
46    /// If it is desired to draw using the position, rotation, etc of the plugin entity
47    /// it is necessary to push this matrix first.
48    pub fn world_matrix(&self) -> Matrix4 {
49        unsafe { (*self.inner).worldMatrix }
50    }
51
52    /// Returns the plugin entity projection matrix.
53    pub fn proj_matrix(&self) -> Matrix4 {
54        unsafe { (*self.inner).projMatrix }
55    }
56
57    pub fn viewport(&self) -> (i32, i32, i32, i32) {
58        unsafe {
59            (
60                (*self.inner).viewportX,
61                (*self.inner).viewportY,
62                (*self.inner).viewportWidth,
63                (*self.inner).viewportHeight,
64            )
65        }
66    }
67
68    pub fn target_size(&self) -> (i32, i32) {
69        unsafe { ((*self.inner).targetWidth, (*self.inner).targetHeight) }
70    }
