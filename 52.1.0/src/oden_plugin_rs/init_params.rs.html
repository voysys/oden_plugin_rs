6/// # InitParams
7/// [`OdenPlugin::init`](crate::OdenPlugin::init) is called when the plugin is initialized.
8/// This is an appropriate place to create data for storing state and/or initializing threads.
9///
10/// InitParams is a struct that exposes the Oden Plugin API functions associated with
11/// the [`OdenPlugin::init`](crate::OdenPlugin::init) function.
12/// The API is passed as a parameter to `init(api: &InitParams)` and can be accessed as shown in the example below:
13///
14/// # Examples
15/// ```ignore
16/// fn init(api: &InitParams) -> Self {
17///     State { _dummy: false }
18/// }
19/// ```
20pub struct InitParams<'a> {
21    #[doc(hidden)]
22    pub inner: *mut crate::plugin_h::OdenPluginEntityInitParams,
23    #[doc(hidden)]
24    pub phantom: PhantomData<&'a i32>,
25}
